#include <tunables/global>

profile layerv_demo_display flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  /usr/local/bin/python rix,
  /usr/local/bin/python3 rix,
  /usr/local/bin/python3.12 rix,
  /usr/local/bin/qurl-connector rcx -> connector,
  /usr/local/lib/** mr,
  # Playwright's Python client starts this fixed, packaged protocol driver.
  /usr/local/lib/python3.12/site-packages/playwright/driver/node rix,
  /app/{,**} r,
  /run.sh rix,
  /usr/lib/chromium/chromium rix,
  /usr/lib/chromium/{,**} mrrix,
  /usr/share/chromium/{,**} r,
  /usr/share/fonts/{,**} r,
  /usr/share/fontconfig/{,**} r,
  /var/cache/fontconfig/{,**} r,
  /etc/fonts/{,**} r,

  /etc/hosts r,
  /etc/hostname r,
  /etc/machine-id r,
  /etc/nsswitch.conf r,
  /etc/passwd r,
  /etc/group r,
  /etc/resolv.conf r,
  /etc/ssl/openssl.cnf r,
  /etc/ssl/certs/{,**} r,
  /var/lib/dbus/machine-id r,

  /data/options.json r,
  /data/secrets/{,**} rwk,
  /data/connector-config/{,**} rwk,
  /data/connector-state/{,**} rwk,
  /data/connector-logs/{,**} rwk,
  /tmp/{,**} rwk,
  /dev/shm/{,**} rwk,
  /proc/{,**} r,
  /sys/devices/system/cpu/** r,
  /dev/null rw,
  /dev/urandom r,

  network inet stream,
  network inet6 stream,
  network inet dgram,
  network inet6 dgram,
  # Chromium's browser, renderer, and utility processes communicate only
  # inside the container over Unix-domain IPC and same-profile signals.
  network unix stream,
  network unix dgram,

  # The fixed root bootstrap reads Supervisor-owned options, then permanently
  # drops to the dedicated runtime identity before binding the HTTP listener.
  capability setuid,
  capability setgid,
  capability chown,

  signal (send, receive) peer=layerv_demo_display,
  signal (send, receive) peer=layerv_demo_display//connector,
  signal (receive) set=(int, term),

  # The connector has a separate execution domain. It cannot read the App's
  # Python source, Chromium profile, HA token, frames, or viewer memory.
  profile connector {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    /usr/local/bin/qurl-connector mr,

    /etc/hosts r,
    /etc/hostname r,
    /etc/machine-id r,
    /etc/nsswitch.conf r,
    /etc/passwd r,
    /etc/group r,
    /etc/resolv.conf r,
    /etc/ssl/openssl.cnf r,
    /etc/ssl/certs/{,**} r,
    /var/lib/dbus/machine-id r,

    /data/secrets/layerv-api-key r,
    /data/connector-config/{,**} rwk,
    /data/connector-state/{,**} rwk,
    /data/connector-logs/{,**} rwk,
    /tmp/{,**} rwk,
    /proc/{,**} r,
    /sys/devices/system/cpu/** r,
    /dev/null rw,
    /dev/urandom r,

    network inet stream,
    network inet6 stream,
    network inet dgram,
    network inet6 dgram,
    network unix stream,
    network unix dgram,

    signal (send, receive) peer=layerv_demo_display,
    signal (send, receive) peer=layerv_demo_display//connector,
  }
}
